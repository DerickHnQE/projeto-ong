/*
 * js/main.js
 * Este script aplica máscaras de input nos campos do formulário de cadastro.
 */
document.addEventListener('DOMContentLoaded', () => {

    // Função genérica para aplicar máscara
    // Para este projeto, usaremos funções específicas para clareza.

    const cpfInput = document.getElementById('cpf');
    const telefoneInput = document.getElementById('telefone');
    const cepInput = document.getElementById('cep');

    if (cpfInput) {
        cpfInput.addEventListener('input', (e) => {
            let value = e.target.value.replace(/\D/g, ''); // Remove tudo que não é dígito
            value = value.replace(/(\d{3})(\d)/, '$1.$2'); // Coloca o primeiro ponto
            value = value.replace(/(\d{3})(\d)/, '$1.$2'); // Coloca o segundo ponto
            value = value.replace(/(\d{3})(\d{1,2})$/, '$1-$2'); // Coloca o traço
            e.target.value = value.slice(0, 14); // Limita ao tamanho máximo
        });
    }

    if (telefoneInput) {
        telefoneInput.addEventListener('input', (e) => {
            let value = e.target.value.replace(/\D/g, '');
            value = value.replace(/^(\d{2})(\d)/, '($1) $2'); // Coloca os parênteses e espaço
            value = value.replace(/(\d{5})(\d)/, '$1-$2'); // Coloca o traço
            e.target.value = value.slice(0, 15); // Limita ao tamanho (XX) XXXXX-XXXX
        });
    }

    if (cepInput) {
        cepInput.addEventListener('input', (e) => {
            let value = e.target.value.replace(/\D/g, '');
            value = value.replace(/(\d{5})(\d)/, '$1-$2'); // Coloca o traço
            e.target.value = value.slice(0, 9); // Limita ao tamanho XXXXX-XXX
        });
    }
    
    // Opcional: Impedir o envio do formulário para demonstração
    const form = document.getElementById('cadastroForm');
    if (form) {
        form.addEventListener('submit', (e) => {
            e.preventDefault(); // Impede o envio real
            console.log('Formulário enviado (simulação)!');
            // Aqui você poderia adicionar uma mensagem de sucesso na tela
            alert('Cadastro enviado com sucesso! (Esta é uma simulação)');
            form.reset(); // Limpa o formulário
        });
    }

});
